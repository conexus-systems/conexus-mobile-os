cmake_minimum_required(VERSION 3.16)
project(conexus-shell-poc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick Gui Test)
qt_add_executable(conexus-shell src/main.cpp)
qt_add_qml_module(conexus-shell URI Conexus.Shell VERSION 1.0 RESOURCE_PREFIX "/"
  QML_FILES qml/Main.qml qml/ConexusTile.qml)
target_link_libraries(conexus-shell PRIVATE Qt6::Quick Qt6::Gui)
qt_add_executable(tst_shell tests/tst_shell.cpp)
target_compile_definitions(tst_shell PRIVATE SRC_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tst_shell PRIVATE Qt6::Quick Qt6::Gui Qt6::Test Qt6::Qml)
enable_testing()
add_test(NAME shell_states COMMAND tst_shell)
